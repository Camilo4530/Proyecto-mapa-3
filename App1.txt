@app.route('/editar_grafo', methods=['POST'])
def editar_grafo():
    accion = request.form['accion']
    global nodos_agregados  # Asegurar que modificamos la lista global

    if accion == 'agregar_nodo':
        direccion = request.form['direccion']
        try:
            lat, lon = ox.geocode(direccion)
            nuevo_id = max(G.nodes) + 1
            G.add_node(nuevo_id, x=lon, y=lat)
            nodos_agregados.append({'id': nuevo_id, 'direccion': direccion})
        except Exception as e:
            print(f"Error al agregar nodo: {e}")

    elif accion == 'eliminar_nodo':
        nodo_id = int(request.form['nodo'])
        if nodo_id in G.nodes:
            G.remove_node(nodo_id)
            nodos_agregados = [nodo for nodo in nodos_agregados if nodo['id'] != nodo_id]

    return render_template('index.html', nodos=nodos_agregados)

def buscar_direcciones():
    texto = request.args.get('q', '')
    if not texto:
        return jsonify([])
    
    try:
        lugares = ox.geocode_to_gdf(texto + ", Bogotá, Colombia")
        sugerencias = lugares['display_name'].tolist()
        return jsonify(sugerencias[:5])  # Solo las 5 primeras sugerencias
    except Exception as e:
        print(f"Error en autocompletar: {e}")
        return jsonify([])
